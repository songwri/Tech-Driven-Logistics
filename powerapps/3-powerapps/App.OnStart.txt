// [PA 전용] 로그인한 사람의 회사 계정 정보 (신청 담당자 칸을 미리 채움). 웹은 직접 입력
Set(gMe, Office365Users.MyProfileV2());

// [PA 전용] 관리자 여부를 회사 로그인으로 판단 (웹은 관리자 키 입력). TourAdmins 목록에 있으면 "main" · "mtv" · "hq", 없으면 빈 값
// 시화MTV 를 닫아 두면(MtvEnabled = false) 시화MTV · 통합 관제 관리자는 아직 들어올 수 없다 (웹의 '준비 중' 과 같음)
Set(gAdminSite, With({site: LookUp(TourAdmins, Title = Lower(User().Email), Site)}, If(MtvEnabled || site = "main", site, Blank())));

// 신청 화면 상태
Set(gStep, 1);
Set(gTour, First(Tours));
Set(gMonth, Date(Year(EarliestBase), Month(EarliestBase), 1));
Set(gDate, Blank());
Set(gSlot, Blank());
Set(gCategory, "external");
Set(gClientType, "");
Set(gLang, "ko");
Set(gNeedReset, false);
Set(gShowBlock, false);
Set(gAdminFilter, "대기");
Set(gSel, Blank());
Set(gBlockTarget, "메가와이즈");
Set(gBlockTime, "종일");

// 업종 · 방문 목적: 통계를 위해 하나씩만 고른다
Set(gIndustry, "");
Set(gPurpose, "");

// 통계 화면: 연간 · 월별 · 기간 지정 · 전체 기간, 집계 기준(전체 / 승인 · 완료)
Set(gStatPeriod, "year");
Set(gStatYear, Year(Today()));
Set(gStatMonth, Month(Today()));
Set(gStatBasis, "all");
Set(gStatTab, "overview");

// 관리자 예약 달력
Set(gCalMonth, Date(Year(Today()), Month(Today()), 1));
Set(gCalDay, Text(Today(), "yyyy-mm-dd"));
Set(gCalTour, "");
Set(gOpenId, Blank());
Set(gBlockDate, Blank());

// 이번 달 예약 점유 · 막힌 일정 (개인정보 없음)
ClearCollect(colBusy, Filter(TourBusy, MonthKey = Text(gMonth, "yyyy-mm")))
