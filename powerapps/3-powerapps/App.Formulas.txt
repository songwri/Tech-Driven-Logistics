// 시화MTV 가 들어간 투어: 3영업일 뒤 (주말 · 공휴일 제외)와 기본 중 더 늦은 날
EarliestMtv = With(
    {
        d: Last(
            FirstN(
                Filter(
                    ForAll(Sequence(20), {D: DateAdd(Today(), Value, TimeUnit.Days)}),
                    Weekday(D, StartOfWeek.Monday) <= 5 && !(Text(D, "yyyy-mm-dd") in HolidayKeys.Value)
                ),
                3
            )
        ).D
    },
    If(d > EarliestBase, d, EarliestBase)
);
